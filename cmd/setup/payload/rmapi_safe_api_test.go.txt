package api

import (
	"encoding/base64"
	"fmt"
	"strings"
	"testing"
	"time"
)

func safeTestToken(payload string) string {
	header := base64.RawURLEncoding.EncodeToString([]byte(`{"alg":"none"}`))
	body := base64.RawURLEncoding.EncodeToString([]byte(payload))
	return header + "." + body + ".signature"
}

func TestSafeParseTokenAcceptsBoundedCurrentClaims(t *testing.T) {
	token := safeTestToken(fmt.Sprintf(`{"auth0-profile":{"email":"person@example.test"},"scopes":"sync:hare","exp":%d}`, time.Now().Add(time.Hour).Unix()))
	info, err := ParseToken(token)
	if err != nil {
		t.Fatal(err)
	}
	if info.User != "person@example.test" || info.SyncVersion != Version15 {
		t.Fatalf("unexpected token information: %#v", info)
	}
}

func TestSafeParseTokenRejectsExpiredClaims(t *testing.T) {
	token := safeTestToken(fmt.Sprintf(`{"exp":%d}`, time.Now().Add(-time.Hour).Unix()))
	if _, err := ParseToken(token); err == nil || !strings.Contains(err.Error(), "expired") {
		t.Fatalf("unexpected error: %v", err)
	}
}

func TestSafeParseTokenRejectsMalformedClaims(t *testing.T) {
	token := safeTestToken(`{"exp":"not-a-number"}`)
	if _, err := ParseToken(token); err == nil {
		t.Fatal("malformed claims were accepted")
	}
}

func TestSafeParseTokenRejectsOversizedInput(t *testing.T) {
	if _, err := ParseToken(strings.Repeat("x", maxUserTokenBytes+1)); err == nil || !strings.Contains(err.Error(), "size") {
		t.Fatalf("unexpected error: %v", err)
	}
}
